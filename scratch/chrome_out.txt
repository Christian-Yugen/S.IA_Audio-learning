async function testAudioGraph() {
  const sampleRate = 22050;
  const duration = 0.5; // seconds
  const length = sampleRate * duration;

  async function renderAngle(angle, distance) {
    const ctx = new OfflineAudioContext(2, length, sampleRate);
    
    // Create a mono test beep (440Hz sine wave)
    const buffer = ctx.createBuffer(1, length, sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < length; i++) {
      data[i] = Math.sin(2 * Math.PI * 440 * (i / sampleRate));
    }

    const source = ctx.createBufferSource();
    source.buffer = buffer;
    const gain = ctx.createGain();
    gain.gain.value = 1.0;
    source.connect(gain);

    // Build stage
    const input = ctx.createGain();
    const output = ctx.createGain();
    output.connect(ctx.destination);
    gain.connect(input);

    const bypassGain = ctx.createGain();
    bypassGain.gain.value = 0.0;
    input.connect(bypassGain);
    bypassGain.connect(output);

    const distanceFilter = ctx.createBiquadFilter();
    distanceFilter.type = "lowpass";
    distanceFilter.frequency.value = 8000;

    const clarityFilter = ctx.createBiquadFilter();
    clarityFilter.type = "peaking";
    clarityFilter.frequency.value = 3800;

    const leftGain = ctx.createGain();
    const rightGain = ctx.createGain();
    const leftDelay = ctx.createDelay(0.01);
    const rightDelay = ctx.createDelay(0.01);
    const merger = ctx.createChannelMerger(2);

    const distanceGain = ctx.createGain();
    const spatialGain = ctx.createGain();
    spatialGain.gain.value = 1.15;

    // Connect mono chain
    input.connect(distanceFilter);
    distanceFilter.connect(clarityFilter);

    // Left branch
    clarityFilter.connect(leftDelay);
    leftDelay.connect(leftGain);
    leftGain.connect(merger, 0, 0);

    // Right branch
    clarityFilter.connect(rightDelay);
    rightDelay.connect(rightGain);
    rightGain.connect(merger, 0, 1);

    // Post merger
    merger.connect(distanceGain);
    distanceGain.connect(spatialGain);
    spatialGain.connect(output);

    // Apply angles
    const u = Math.max(-1.0, Math.min(1.0, angle / 90));
    let lGain, rGain;
    if (u <= 0) {
      lGain = 1.0;
      rGain = Math.pow(Math.max(0, 1.0 + u), 1.5);
    } else {
      lGain = Math.pow(Math.max(0, 1.0 - u), 1.5);
      rGain = 1.0;
    }
    leftGain.gain.value = lGain;
    rightGain.gain.value = rGain;
    leftDelay.delayTime.value = u > 0 ? u * 0.0006 : 0;
    rightDelay.delayTime.value = u < 0 ? Math.abs(u) * 0.0006 : 0;

    source.start(0);
    const renderedBuffer = await ctx.startRendering();

    const leftChannel = renderedBuffer.getChannelData(0);
    const rightChannel = renderedBuffer.getChannelData(1);

    // Calculate RMS
    let sumL = 0, sumR = 0;
    for (let i = 0; i < length; i++) {
      sumL += leftChannel[i] * leftChannel[i];
      sumR += rightChannel[i] * rightChannel[i];
    }
    const rmsL = Math.sqrt(sumL / length);
    const rmsR = Math.sqrt(sumR / length);

    return {
      angle,
      rmsL: Number(rmsL.toFixed(4)),
      rmsR: Number(rmsR.toFixed(4)),
      ratioLR: rmsR === 0 ? "INF" : Number((rmsL / rmsR).toFixed(2))
    };
  }

  const results = [];
  results.push(await renderAngle(-90, 1.0));
  results.push(await renderAngle(-60, 1.0));
  results.push(await renderAngle(0, 1.0));
  results.push(await renderAngle(60, 1.0));
  results.push(await renderAngle(90, 1.0));

  document.getElementById("output").textContent = JSON.stringify(results, null, 2);
}

testAudioGraph();
</script>
